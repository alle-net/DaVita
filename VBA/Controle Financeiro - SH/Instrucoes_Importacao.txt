- Cole o codigo abaixo SUBSTITUINDO tudo:

---------- COPIAR ----------
Option Explicit

Private Sub Workbook_Open()
    modAutenticacao.ResetarSessao
    frmLogin.Show vbModal
    If modAutenticacao.UsuarioAtual > 0 Then
        If modDados.CarregarDadosUsuario(modAutenticacao.UsuarioAtual) Then
            frmPrincipal.Show vbModeless
        End If
    End If
End Sub

Private Sub Workbook_BeforeClose(Cancel As Boolean)
    On Error Resume Next
    Sheets("Dados").AutoFilterMode = False
    modAutenticacao.ResetarSessao
End Sub
