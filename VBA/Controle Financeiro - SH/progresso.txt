8. CODIGO ATUAL DO frmLogin (fornecido ao usuario):
```
Option Explicit

Private Sub UserForm_Initialize()
    CarregarEmails
    Me.Left = 300
    Me.Top = 200
    txBoxSenha.SetFocus
End Sub

Private Sub CarregarEmails()
    Dim em() As String, i As Long
    em = modAutenticacao.CarregarEmailsAtivos
    cmbBoxEmail.Clear
    If (Not em) = -1 Then Exit Sub
    For i = LBound(em) To UBound(em)
        cmbBoxEmail.AddItem em(i)
    Next i
    If cmbBoxEmail.ListCount > 0 Then cmbBoxEmail.ListIndex = 0
End Sub

Private Sub cmbBoxEmail_Change()
    lblMensagem = ""
End Sub

Private Sub txBoxSenha_Change()
    lblMensagem = ""
End Sub

Private Sub cmdButtonEntrar_Click()
    Dim e As String, s As String
    e = Trim(cmbBoxEmail.Value)
    s = Trim(txBoxSenha.Value)
    lblMensagem = ""
    
    If e = "" Then
        lblMensagem = "Selecione um email."
        cmbBoxEmail.SetFocus
        Exit Sub
    End If
    
    If s = "" Then
        lblMensagem = "Digite a senha."
        txBoxSenha.SetFocus
        Exit Sub
    End If
    
    If modAutenticacao.VerificarCredenciais(e, s) Then
        modAutenticacao.UsuarioAtual = modAutenticacao.ObterIdUsuario(e)
        modAutenticacao.EmailAtual = e
        txBoxSenha.Value = ""
        Me.Hide
    Else
        lblMensagem = "Email ou senha invalidos."
        txBoxSenha.Value = ""
        txBoxSenha.SetFocus
    End If
End Sub

Private Sub cmdButtonCancelar_Click()
    ThisWorkbook.Close False
End Sub

Private Sub UserForm_QueryClose(Cancel As Integer, CloseMode As Integer)
    If CloseMode = 0 Then Cancel = 1: ThisWorkbook.Close False
End Sub
```
